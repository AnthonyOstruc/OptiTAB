\begin{pisteaudio}{5.07}{Le pluriel des noms (p.~\pageref{piste-5.07})}
\annonce{Listen and repeat.}
\repetepaire{A pen.}{Two pens.}
\repetepaire{A box.}{Two boxes.}
\repetepaire{A city.}{Two cities.}
\repetepaire{A man.}{Two men.}
\repetepaire{A woman.}{Two women.}
\repetepaire{A child.}{Two children.}
\end{pisteaudio}

\begin{pisteaudio}{5.08}{\en{this}, \en{that}, \en{these}, \en{those} (p.~\pageref{piste-5.08})}
\annonce{Listen and repeat.}
\repetepaire{What's this?}{It's a pen.}
\repetepaire{What's that?}{It's a lamp.}
\repetepaire{What are these?}{They're keys.}
\repetepaire{What are those?}{They're plants.}
\repetepaire{Is this your bag?}{Yes, it is.}
\repetepaire{Are those your books?}{No, they aren't.}
\end{pisteaudio}

\begin{transcription}{5.11}{Exercice 8 · C'est à qui ?}
\repere{1}
\dit{lea}{Jack, is this your phone?}{}
\dit{jack}{No, it isn't. It's Emma's. Her phone is pink.}{}
\repere{2}
\dit{emma}{Léa, are these your keys?}{}
\dit{lea}{Yes, they are! My keys are green. Thank you!}{}
\repere{3}
\dit{jack}{Is that Mr.\ Carter's umbrella?}{}
\dit{emma}{Yes, it is. His umbrella is black.}{}
\repere{4}
\dit{sam}{Léa, is this your bag?}{}
\dit{lea}{No, it isn't. It's Maya's. Her bag is purple.}{}
\end{transcription}

\begin{transcription}{5.12}{Exercice 9 · Dictée à trous}
\dit{sam}{What's this?}{}

\dit{sam}{And what are these?}{}
\dit{lea}{They're my books.}{}
\dit{sam}{Are those your plants?}{}
\dit{lea}{Yes, they are. They're from Mrs.\ Lopez.}{}
\dit{sam}{The lamp is orange!}{}
\dit{lea}{Yes, and all my boxes are brown!}{}
\end{transcription}
