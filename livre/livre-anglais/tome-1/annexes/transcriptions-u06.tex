\corrigesUnite{6}{A day in my life}

\begin{pisteaudio}{6.07}{Le présent simple (p.~\pageref{piste-6.07})}
\annonce{Listen and repeat.}
\repetepaire{I work in New York.}{He works in New York.}
\repetepaire{I get up at six.}{She gets up at six.}
\repetepaire{I go to the gym.}{He goes to the gym.}
\repetepaire{I watch TV.}{She watches TV.}
\repetepaire{I study music.}{He studies music.}
\repetepaire{I have breakfast.}{She has breakfast.}
\end{pisteaudio}

\begin{pisteaudio}{6.08}{Dire l'heure (p.~\pageref{piste-6.08})}
\annonce{Listen and repeat.}
\repete{It's nine o'clock.}
\repetepaire{It's nine fifteen.}{It's a quarter past nine.}
\repetepaire{It's nine thirty.}{It's half past nine.}
\repetepaire{It's nine forty-five.}{It's a quarter to ten.}
\repete{It's twelve o'clock. It's noon.}
\repete{It's seven p.m.}
\end{pisteaudio}

\begin{transcription}{6.11}{Exercice 8 · La journée d'Emma}
