\planPartie{1}{Hello, Python}{Your first programs}{otNavy}{1}{4}
  {Install Python, talk to it in the shell and write your first programs. You will learn to
   calculate, to remember values with variables and to work with text.}
\planPartie{2}{Listen and decide}{Interactive programs}{otTeal}{5}{8}
  {Your programs start a conversation: they ask questions, check the answers and react
   differently depending on the situation.}
\planPartie{3}{Loops}{Repeat without repeating yourself}{otOrangeDark}{9}{12}
  {Computers never get tired of repeating. Make your programs repeat actions, walk through
   sequences and run simple games.}
\planPartie{4}{Functions}{Organise your code}{otViolet}{13}{16}
  {Group instructions into reusable functions, give them parameters and return values, and make
   your programs robust when something goes wrong.}
\planPartie{5}{Collections and files}{Handle lots of data}{otGreenDark}{17}{20}
  {Store many values in lists and dictionaries, process them with loops, and save your data in
   files so that it survives when the program stops.}

%% \planChapitre{n}{titre}{sous-titre}{concepts}{python}{bonnes habitudes}{réalisation}
\planChapitre{1}{Hello, Python!}{Install Python and run your first programs}
  {program, instruction, interpreter, shell and script, error message}
  {\py{print()}, text in quotes, comments with \py{\#}}
  {read error messages from the bottom up; save and run often}
  {a business card printed by Python}
\planChapitre{2}{Python as a calculator}{Numbers, operators and expressions}
  {expression, integers and decimals, order of operations}
  {\py{+ - * /}, \py{// \% **}, \py{round()}, \py{abs()}, \py{min()}, \py{max()}}
  {spaces around operators; parentheses for clarity}
  {a holiday budget calculator}
\planChapitre{3}{Variables}{Give names to your values}
  {variable, assignment, names and objects, types}
  {\py{=}, \py{+=}, \py{-=}, \py{type()}, multiple assignment}
  {meaningful names in \py{snake\_case}; constants in capitals}
  {a holiday planner you can update in one line}
\planChapitre{4}{Working with text}{Strings, f-strings and special characters}
  {string, concatenation, length, escape sequence}
  {\py{+} and \py{*} on text, \py{len()}, f-strings, \py{\textbackslash n}, \py{str()}}
  {f-strings rather than long concatenations}
  {a poster generator}
\planChapitre{5}{Talking with the user}{\py{input()} and type conversions}
  {user input, converting text to numbers}
  {\py{input()}, \py{int()}, \py{float()}, \py{str()}}
  {clear prompts; convert once, as early as possible}
  {a currency converter}
\planChapitre{6}{More about strings}{Indexing, slicing and string methods}
  {index, slice, method, immutable value}
  {\py{s[i]}, \py{s[a:b]}, \py{.upper()}, \py{.strip()}, \py{.replace()}, \py{in}}
  {use built-in methods instead of reinventing them}
  {a username generator}
\planChapitre{7}{True or false?}{Booleans, comparisons and logic}
  {boolean, comparison, logical operators}
  {\py{True}, \py{False}, \py{== != < >}, \py{and}, \py{or}, \py{not}}
  {readable conditions; never \py{== True}}
  {an eligibility checker}
\planChapitre{8}{Making decisions}{\py{if}, \py{elif} and \py{else}}
  {condition, block, indentation, nested decisions}
  {\py{if}, \py{elif}, \py{else}}
  {four-space indentation; flat rather than deeply nested}
  {a grade calculator}
\planChapitre{9}{Repeating with \py{while}}{Loops that run while a condition is true}
  {loop, iteration, stopping condition, infinite loop}
  {\py{while}, counters, input checking}
  {make sure every loop can end}
  {a password checker with three attempts}
\planChapitre{10}{Counting with \py{for}}{\py{for} loops and \py{range()}}
  {iterating over a sequence, ranges of numbers}
  {\py{for}, \py{range()}, looping over text}
  {\py{for} when you know how many times}
  {multiplication tables}
\planChapitre{11}{Loop recipes}{Accumulate, count, search and stop}
  {accumulator, counter, flag, nested loops}
  {\py{break}, \py{continue}, nested \py{for}}
  {one loop, one job}
  {a pattern drawer}
\planChapitre{12}{Randomness and games}{The \py{random} module}
  {module, import, pseudo-random numbers}
  {\py{import}, \py{randint()}, \py{choice()}, \py{random()}}
  {imports at the top of the file}
  {a dice simulator}
\planChapitre{13}{Your own functions}{\py{def}, calls and parameters}
  {function, definition and call, parameter, argument}
  {\py{def}, parameters, calls}
  {one function, one task; verb names}
  {a greeting-card generator}
\planChapitre{14}{Return values}{Functions that give back a result}
  {return value, \py{None}, composition}
  {\py{return}, returning several values}
  {return rather than print}
  {a geometry toolkit}
\planChapitre{15}{Functions like a pro}{Scope, default values and documentation}
  {local and global scope, docstring, testing}
  {default values, keyword arguments, docstrings, \py{assert}}
  {document and test every function}
  {a tested password-strength checker}
\planChapitre{16}{When things go wrong}{Exceptions and \py{try}/\py{except}}
  {exception, traceback, error handling}
  {\py{try}, \py{except}, \py{else}, \py{finally}}
  {catch specific exceptions only}
  {a crash-proof calculator}
\planChapitre{17}{Lists}{Storing many values in order}
  {list, index, mutable value}
  {\py{[ ]}, \py{append()}, \py{insert()}, \py{remove()}, \py{pop()}, \py{len()}}
  {plural names for lists}
  {a shopping list}
\planChapitre{18}{Working with lists}{Loops, sorting and comprehensions}
  {traversal, sorting, lists of lists, comprehension}
  {\py{sorted()}, \py{.sort()}, \py{sum()}, comprehensions}
  {never change a list while looping over it}
  {a class gradebook}
\planChapitre{19}{Dictionaries and tuples}{Look up values by key}
  {key-value pair, tuple, unpacking}
  {\py{\{ \}}, \py{d[key]}, \py{.get()}, \py{.items()}, tuples}
  {choose the right collection}
  {a contact book}
\planChapitre{20}{Files}{Save and load your data}
  {file, path, reading and writing, CSV}
  {\py{open()}, \py{with}, \py{read()}, \py{write()}, \py{split()}}
  {always use \py{with}; expect missing files}
  {a diary that remembers}
