\corrigesUnite{11}{Can I have a coffee, please?}

\begin{pisteaudio}{11.07}{Demander poliment (p.~\pageref{piste-11.07})}
\annonce{Listen and repeat.}
\repete{Can I have a coffee, please?}
\repete{I'd like a coffee, please.}
\repete{I'll have a coffee, please.}
\repete{Could I have a coffee, please?}
\repete{Can we have two teas, please?}
\repete{Can we have the check, please?}
\end{pisteaudio}

\begin{pisteaudio}{11.08}{Proposer et répondre (p.~\pageref{piste-11.08})}
\annonce{Listen and repeat.}
\repetepaire{Would you like a muffin?}{Yes, please.}
\repetepaire{Would you like some tea?}{No, thank you.}
\repetepaire{Would you like something to drink?}{Yes, a glass of water, please.}
\repetepaire{Would you like dessert?}{No, thanks.}
\end{pisteaudio}

\begin{transcription}{11.10}{Exercice 7 · Les formes faibles}
\annonce{Listen and complete.}
\numdire{1}{Can I have a tea, please? … Can I have a tea, please?}
\numdire{2}{Could I have the check? … Could I have the check?}
\numdire{3}{Would you like some water? … Would you like some water?}
\numdire{4}{I'd like a salad, please. … I'd like a salad, please.}
\numdire{5}{Can we have two coffees? … Can we have two coffees?}
\numdire{6}{I'll have the steak. … I'll have the steak.}
\end{transcription}

\begin{transcription}{11.11}{Exercice 8 · Les commandes}
\repere{1}
\dit{sam}{Hi, Emma! What can I get you?}{}
\dit{emma}{A small tea, please.}{}
\dit{sam}{For here or to go?}{}
\dit{emma}{For here, please.}{}
\repere{2}
\dit{jack}{A large coffee to go, please. I'm late!}{}
\dit{sam}{Again, Jack?}{}
\repere{3}
\dit{maya}{Hi, Sam! Can I have a medium latte? To go, please.}{}
\dit{sam}{Sure, little sister!}{}
\end{transcription}

\begin{transcription}{11.12}{Exercice 9 · Dictée à trous}
\dit{serveur}{Are you ready to order?}{}
\dit{emma}{Yes. I'd like the soup, please.}{}
\dit{serveur}{Would you like something to drink?}{}
\dit{emma}{Yes, please. Can I have a glass of orange juice?}{}
\dit{serveur}{Of course. And for dessert?}{}
\dit{emma}{No, thank you.}{}
\end{transcription}

\begin{transcription}{11.13}{Exercice 12 · À vous de parler !}

\silence{3}
\dit{sam}{For here or to go?}{}
\silence{3}
\dit{sam}{Would you like a muffin with that?}{}
\silence{4}
\dit{sam}{That's six dollars, please.}{}

\end{transcription}
