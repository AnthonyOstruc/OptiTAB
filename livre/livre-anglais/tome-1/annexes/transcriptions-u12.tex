\corrigesUnite{12}{I can swim!}

\begin{pisteaudio}{12.07}{\en{can} et \en{can't} (p.~\pageref{piste-12.07})}
\annonce{Listen and repeat.}
\repete{I can swim.}
\repete{She can sing.}
\repete{He can't ski.}
\repete{We can't come.}
\repete{They can play the piano.}
\repete{You can drive.}
\end{pisteaudio}

\begin{pisteaudio}{12.08}{Questions et réponses courtes (p.~\pageref{piste-12.08})}
\annonce{Listen and repeat.}
\repetepaire{Can you swim?}{Yes, I can. I swim very well.}
\repetepaire{Can she ski?}{No, she can't.}
\repetepaire{Can they dance?}{Yes, they can. They dance quite well.}
\repetepaire{Can he cook?}{No, he can't cook at all.}
\end{pisteaudio}

\begin{transcription}{12.10}{Exercice 7 · \en{can} ou \en{can't} ?}
\annonce{Listen. Can or can't?}
\numdire{1}{Léa can ski. … Léa can ski.}
\numdire{2}{Jack can't cook. … Jack can't cook.}
\numdire{3}{Emma can't swim. … Emma can't swim.}
\numdire{4}{Sam can sing. … Sam can sing.}
\numdire{5}{Maya can't drive. … Maya can't drive.}
\numdire{6}{Mr.\ Carter can dance. … Mr.\ Carter can dance.}
\end{transcription}

\begin{transcription}{12.11}{Exercice 8 · Les talents de Maya}
\dit{sam}{Maya, can you sing?}{}
\dit{maya}{Yes, I can! I sing quite well.}{}
\dit{sam}{Can you play the piano?}{}
\dit{maya}{No, I can't. But I can dance!}{}
\dit{sam}{Can you drive a car?}{}
\dit{maya}{No, I can't. I'm twenty and I can't drive!}{}
\dit{sam}{And can you speak Spanish?}{}
\dit{maya}{Yes, I can. Our grandmother is from Puerto Rico!}{}
\end{transcription}

\begin{transcription}{12.12}{Exercice 9 · Dictée à trous}
\dit{coach}{Can you play tennis, Emma?}{}
\dit{emma}{Yes, I can. I play quite well.}{}
\dit{coach}{And can you swim?}{}
\dit{emma}{No, I can't. I can't swim at all!}{}
\dit{coach}{No problem. Can you run?}{}
\dit{emma}{Oh yes! I run very fast!}{}
\end{transcription}

\begin{transcription}{12.13}{Exercice 12 · À vous de parler !}
