\begin{transcription}{2.12}{Exercice 9 · Dictée à trous}
\dit{sam}{Hi, Léa! Are you OK?}{}
\dit{lea}{Yes, I'm fine, thanks.}{}
\dit{sam}{Are you from Paris?}{}
\dit{lea}{No, I'm not. I'm from Lyon.}{}
\dit{sam}{Is Lyon a big city?}{}
\dit{lea}{Yes, it is. And you? Where are you from?}{}
\dit{sam}{I'm from here, from Brooklyn!}{}
\dit{lea}{Really? Great!}{}
\end{transcription}
