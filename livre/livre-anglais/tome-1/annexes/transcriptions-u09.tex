\begin{transcription}{9.12}{Exercice 9 · Dictée à trous}
\dit{passant}{Excuse me, how do I get to the station?}{}
\dit{jack}{Go straight ahead, then turn right.}{}
\dit{passant}{Is it far?}{}
\dit{jack}{No, it's a ten-minute walk. It's opposite the museum.}{}
\dit{passant}{Can I take the bus?}{}
\dit{jack}{Yes, but don't take the number 4. Take the number 2.}{}
\end{transcription}
