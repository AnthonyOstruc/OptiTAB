\corrigesUnite{17}{Where were you?}

\begin{pisteaudio}{17.07}{\en{was} et \en{were} (p.~\pageref{piste-17.07})}
\annonce{Listen and repeat.}
\repete{I was at home.}
\repete{She was very tired.}
\repete{We were in London.}
\repete{They weren't late.}
\repete{It was a great party!}
\repete{There were a lot of people.}
\end{pisteaudio}

\begin{pisteaudio}{17.08}{Questions au passé (p.~\pageref{piste-17.08})}

\repetepaire{Was it cold?}{No, it wasn't.}

\end{pisteaudio}

\begin{transcription}{17.11}{Exercice 8 · Le mystère du gâteau}
\dit{carter}{Where's my chocolate cake? It was in the kitchen this morning!}{}
\dit{carter}{Jack, where were you at three o'clock?}{}
\dit{jack}{Me? I was at the post office.}{}
\dit{carter}{And you, Emma?}{}
\dit{emma}{I was in the meeting room, with a client.}{}
\dit{carter}{Léa?}{}
\dit{lea}{I was at my desk. And Jack was with me!}{}
\dit{carter}{Jack! You weren't at the post office!}{}
\dit{jack}{Well… the cake was delicious!}{}
\end{transcription}

\begin{transcription}{17.12}{Exercice 9 · Dictée à trous}
\dit{lopez}{Hi, Léa! How was Christmas?}{}
\dit{lea}{It was wonderful! My parents were here for two weeks.}{}
\dit{lopez}{Were they happy in New York?}{}
\dit{lea}{Yes, they were. But it was very cold!}{}
\dit{lopez}{And where were you on New Year's Eve?}{}
\dit{lea}{In Times Square. There were so many people!}{}
\dit{lopez}{Ha! I was there fifty years ago!}{}
\end{transcription}
