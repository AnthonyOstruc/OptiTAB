% Établissement du décalage Doppler : deux maximums successifs émis par une source mobile (L'essentiel, chapitre 23)
\begin{tikzpicture}[x=1cm,y=1cm]
\draw[otMuted,line width=0.4pt,-{Stealth[length=1.5mm]}] (-0.6,0) -- (9.6,0);
\fill[otInk] (9.0,0) circle (0.06) node[pcLegende,anchor=north] {récepteur};
\fill[otRed!80!black] (0,0) circle (0.08) node[pcLegende,anchor=north,text=otRed!80!black] {$S$ à $t=0$};
\fill[otRed!50] (2.0,0) circle (0.08) node[pcLegende,anchor=north,text=otRed!70!black] {$S$ à $t=T_\mathrm{E}$};
\draw[otNavy,line width=0.9pt] (6.0,-0.4) -- (6.0,0.4) node[pcLegende,anchor=south,text=otNavy] {1\textsuperscript{er} maximum};
\draw[otNavy,line width=0.9pt] (2.0,0.12) -- (2.0,0.4) node[pcLegende,anchor=south,text=otNavy] {2\textsuperscript{e} maximum};
\draw[otMuted,line width=0.4pt,{Stealth[length=1.2mm]}-{Stealth[length=1.2mm]}] (0,0.95) -- (6.0,0.95) node[pos=0.5,pcLegende,anchor=south] {$v\,T_\mathrm{E}=\lambda_\mathrm{E}$};
\draw[otMuted,line width=0.4pt,{Stealth[length=1.2mm]}-{Stealth[length=1.2mm]}] (0,-0.8) -- (2.0,-0.8) node[pos=0.5,pcLegende,anchor=north] {$v_\mathrm{E}\,T_\mathrm{E}$};
\draw[otGreenDark,line width=0.5pt,{Stealth[length=1.2mm]}-{Stealth[length=1.2mm]}] (2.0,-1.3) -- (6.0,-1.3) node[pos=0.5,pcLegende,anchor=north,text=otGreenDark] {$\lambda_\mathrm{R}=\lambda_\mathrm{E}-v_\mathrm{E}\,T_\mathrm{E}$};
\end{tikzpicture}
\par\smallskip\legende{À la date $T_\mathrm{E}$, le premier maximum a parcouru $v\,T_\mathrm{E}$ ; la source, qui a avancé de $v_\mathrm{E}\,T_\mathrm{E}$, émet le second : les deux maximums ne sont plus séparés que de $\lambda_\mathrm{R}$ (vitesse de la source exagérée).}
