\corrigesUnite{10}{How much is it?}

\begin{pisteaudio}{10.07}{\en{How much}, \en{how many} (p.~\pageref{piste-10.07})}
\annonce{Listen and repeat.}
\repetepaire{How much is this sweater?}{It's forty dollars.}
\repetepaire{How much are these shoes?}{They're eighty dollars.}
\repetepaire{How many eggs do you need?}{Six eggs, please.}
\repetepaire{How much coffee do you need?}{Two bags.}
\repetepaire{How much money do you have?}{Twenty dollars.}
\repetepaire{How many apples are there?}{There are ten.}
\end{pisteaudio}

\begin{transcription}{10.11}{Exercice 8 · Combien ça coûte ?}
\dit{emma}{Excuse me, how much is this red dress?}{}
\dit{vendeur}{It's forty-five dollars.}{}
\dit{emma}{And these black shoes?}{}
\dit{vendeur}{They're eighty dollars.}{}
\dit{emma}{How much is this T-shirt?}{}
\dit{vendeur}{Thirteen dollars. Not thirty: thirteen!}{}
\dit{emma}{And this winter coat?}{}
\dit{vendeur}{It's ninety-nine dollars. It's on sale!}{}
\dit{emma}{Hmm\ldots{} I'll take the T-shirt!}{}
\end{transcription}

\begin{transcription}{10.12}{Exercice 9 · Dictée à trous}

\end{transcription}
