\corrigesBilan{1}

\begin{transcription}{B1.01}{Exercice 6 · À la fête de Sam}
\dit{sam}{Léa! Hi! Come in!}{}
\dit{lea}{Hi, Sam! This is my friend Emma. Emma, this is Sam, my neighbor.}{}
\dit{sam}{Nice to meet you, Emma.}{}
\dit{emma}{Nice to meet you too.}{}
\dit{sam}{Are you French, like Léa?}{}
\dit{emma}{No, I'm not. I'm British. I'm from London.}{}
\dit{sam}{Cool! And this is my sister, Maya. She's twenty.}{}
\dit{maya}{Hi! Welcome to my brother's party! Is that a British accent?}{}
\dit{emma}{Yes, it is!}{}
\end{transcription}
