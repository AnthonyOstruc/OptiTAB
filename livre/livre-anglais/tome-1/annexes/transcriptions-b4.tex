\corrigesBilan{4}

\begin{transcription}{B4.01}{Exercice 6 · Au téléphone}
\dit{jack}{Hi Léa! What are you doing?}{}
\dit{lea}{Hi Jack! I'm cooking dinner. And you?}{}
\dit{jack}{I'm watching the hockey game on TV. Listen, are you free on Saturday?}{}
\dit{lea}{Yes, I am. Why?}{}
\dit{jack}{Kate and I are going to go ice skating in Central Park. Would you like to come?}{}
\dit{lea}{I'd love to! What time?}{}
\dit{jack}{At two o'clock. But it's going to be very cold: twenty degrees!}{}
\dit{lea}{Twenty degrees Fahrenheit? That's freezing! OK, see you on Saturday!}{}
\end{transcription}
