\corrigesUnite{16}{My plans}

\begin{pisteaudio}{16.07}{\en{be going to} (p.~\pageref{piste-16.07})}

\begin{transcription}{16.11}{Exercice 8 · Les projets de chacun}
\annonce{Listen. What are they going to do for the holidays?}
\repere{First, Emma}
\dit{emma}{For Christmas, I'm going to see my family in London. I can't wait!}{}
\repere{Now, Sam}
\dit{sam}{My band is going to play three concerts in New Jersey.}{}
\repere{Now, Mrs.\ Lopez}
\dit{lopez}{The whole family is here for Christmas. I'm going to cook for twelve people!}{}
\repere{And now, Maya}
\dit{maya}{I'm going to go skiing in Vermont with my friends.}{}
\end{transcription}

\begin{transcription}{16.12}{Exercice 9 · Dictée à trous}
\dit{lopez}{Léa, what are you going to do for Christmas?}{}
\dit{lea}{My parents are going to visit me!}{}
\dit{lopez}{How nice! How long are they going to stay?}{}
\dit{lea}{Two weeks. They're going to fly back on January third.}{}
\dit{lopez}{Are they going to stay with you?}{}
\dit{lea}{No, they aren't. They're going to stay in a hotel.}{}
\end{transcription}
