\begin{transcription}{4.12}{Exercice 9 · Dictée à trous}

\dit{lea}{How old is she?}{}
\dit{sam}{She's twenty. And this is my brother, Ben.}{}
\dit{lea}{Are your parents in Brooklyn too?}{}
\dit{sam}{No, they aren't. They're in Atlanta.}{}
\dit{lea}{And who's this?}{}
\dit{sam}{That's our cat, Mango!}{}
\end{transcription}
