\begin{transcription}{14.11}{Exercice 8 · La météo à la radio}
\dit{radio}{Good morning! Here's the weather for today.}{}
\dit{radio}{In New York, it's cloudy and cool: fifty degrees.}{}
\dit{radio}{In Boston, it's raining: forty-five degrees.}{}
\dit{radio}{In Chicago, it's very windy and cold: thirty-eight degrees.}{}
\dit{radio}{And in Miami, it's sunny and hot: eighty-two degrees! Have a nice day!}{}
\end{transcription}
