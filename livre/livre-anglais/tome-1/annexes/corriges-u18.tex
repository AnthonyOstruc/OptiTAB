\corrigesUnite{18}{What did you do last weekend?}

\begin{corrige}{1}
\begin{reponses}[4]
\item \rep{played}
\item \rep{stopped}
\item \rep{tried}
\item \rep{loved}
\item \rep{studied}
\item \rep{walked}
\item \rep{visited}
\item \rep{planned}
\end{reponses}
\pourquoi{\en{play} : voyelle + \en{y}, on ajoute \en{-ed} ; \en{try}, \en{study} : consonne + \en{y}
→ \en{-ied} ; \en{stop}, \en{plan} : la consonne finale double ; \en{love} : on ajoute seulement \en{-d}.}
\end{corrige}

\begin{corrige}{2}
\begin{reponses}[3]
\item \rep{cleaned}
\item \rep{played}
\item \rep{watched}
\item \rep{didn't}
\item \rep{visited}
\item \rep{cooked}
\end{reponses}
\pourquoi{la croix rouge de l'agenda indique que Jack n'est pas allé au brunch : \en{he didn't go}.}
\end{corrige}

\begin{corrige}{3}
\begin{reponses}[1]
\item \rep{I didn't watch TV.}
\item \rep{She didn't call me.}
\item \rep{We didn't play tennis.}
\item \rep{They didn't visit Boston.}
\item \rep{He didn't study last night.}
\end{reponses}
\piege{après \en{didn't}, le verbe perd son \en{-ed} : \en{She didn't call}, jamais \en{didn't called}.}
\end{corrige}

\begin{corrige}{4}
\begin{reponses}[1]

\item \rep{Did she like the movie?}
\item \rep{What did you cook?}
\item \rep{Did they visit Boston?}
\item \rep{Where did he stay?}
\end{reponses}
\pourquoi{question fermée : \en{Did} + sujet + base verbale ; question ouverte : mot interrogatif +
\en{did} + sujet + base verbale.}
\end{corrige}

\begin{corrige}{11}
Réponse modèle :
\begin{anglais}
\begin{tcolorbox}[otBase,colback=otGreenTint,colframe=otGreenTint,fontupper=\fontsize{10.4}{14}\selectfont]
Last weekend was great! On Saturday morning, I cleaned my apartment. In the afternoon, I played
tennis with my brother. In the evening, I watched a series. On Sunday, I visited my grandmother and
we cooked a big lunch. I didn't work at all!
\end{tcolorbox}
\end{anglais}
\end{corrige}
