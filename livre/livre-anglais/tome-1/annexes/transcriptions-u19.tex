\corrigesUnite{19}{A trip to Boston}

\begin{pisteaudio}{19.07}{Les verbes irréguliers (p.~\pageref{piste-19.07})}

\repete{We went to Boston.}
\repete{I saw the harbor.}
\repete{She bought a cap.}

\repete{I didn't see the game.}
\end{pisteaudio}

\begin{pisteaudio}{19.08}{Questions sur un voyage (p.~\pageref{piste-19.08})}
\annonce{Listen and repeat.}
\repetepaire{Where did you go?}{We went to Boston.}
\repetepaire{How did you travel?}{We took the train.}
\repetepaire{How long did it take?}{About four hours.}
\repetepaire{Did you like it?}{Yes, I did. It was amazing!}
\end{pisteaudio}

\begin{transcription}{19.11}{Exercice 8 · Le récit de Léa}
\dit{lopez}{So, Léa, how was Boston?}{}
\dit{lea}{Wonderful! We left on Saturday at seven in the morning.}{}
\dit{lopez}{Did you take the bus?}{}
\dit{lea}{No, we took the train. It took four hours.}{}
\dit{lopez}{Where did you stay?}{}
\dit{lea}{In a small hotel near the harbor.}{}
\dit{lopez}{And what did you eat?}{}
\dit{lea}{Clam chowder, of course! And lobster!}{}
\end{transcription}

\begin{transcription}{19.12}{Exercice 9 · Dictée à trous}
\dit{sam}{Thanks for the mug! Where did you buy it?}{}
\dit{lea}{I bought it in a small shop near the harbor.}{}
\dit{sam}{Did you see the Old North Church?}{}
\dit{lea}{Yes, we did. Emma took a hundred photos!}{}
\dit{sam}{Ha! And what did you eat?}{}
\dit{lea}{We ate clam chowder. We had a great time!}{}
\end{transcription}
