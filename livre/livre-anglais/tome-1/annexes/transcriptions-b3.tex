\corrigesBilan{3}

\begin{transcription}{B3.01}{Exercice 6 · Au magasin}
\dit{vendeuse}{Hi! Can I help you?}{}
\dit{lea}{Yes, please. I need a warm sweater.}{}
\dit{vendeuse}{What size?}{}
\dit{lea}{Medium.}{}
\dit{vendeuse}{This blue sweater is on sale. It's thirty-nine dollars.}{}
\dit{lea}{It's nice! Can I try it on?}{}
\dit{vendeuse}{Sure! The fitting room is over there.}{}
\dit{lea}{It's perfect. I'll take it. Can I pay by card?}{}
\dit{vendeuse}{Of course!}{}
\end{transcription}
