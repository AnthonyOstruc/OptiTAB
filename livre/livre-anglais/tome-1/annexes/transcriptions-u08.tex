\begin{pisteaudio}{8.07}{\en{there is}, \en{there are} (p.~\pageref{piste-8.07})}

\begin{transcription}{8.11}{Exercice 8 · Les affaires de Jack}
\dit{jack}{Emma, where are my keys?}{}
\dit{emma}{They're on your desk, next to the computer.}{}
\dit{jack}{Oh, thanks. And my umbrella?}{}
\dit{emma}{It's behind the door.}{}
\dit{jack}{And where's my phone?}{}
\dit{emma}{It's under your papers.}{}
\dit{jack}{OK\ldots{} And my coffee? Where's my coffee?}{}
\dit{emma}{It's in your hand, Jack!}{}
\end{transcription}
