\begin{transcription}{18.11}{Exercice 8 · Le week-end de Maya}
\dit{sam}{Hi Maya! Did you have a nice weekend?}{}
\dit{maya}{Yes! On Saturday, I visited a museum with my friend Lily.}{}
\dit{sam}{Cool! Did you like it?}{}
\dit{maya}{I loved it! Then we walked in Central Park.}{}
\dit{sam}{And on Sunday?}{}
\dit{maya}{I stayed home and studied. I have an exam on Tuesday.}{}
\dit{sam}{Oh no! Did you study all day?}{}
\dit{maya}{No, I didn't. In the evening, I watched a series.}{}
\end{transcription}

\begin{transcription}{18.12}{Exercice 9 · Dictée à trous}
\dit{emma}{Did you have a nice weekend, Jack?}{}
\dit{jack}{Yes, I did. I played hockey with Kate.}{}
\dit{emma}{Did you watch the game on Sunday?}{}
\dit{jack}{No, I didn't. I visited my parents.}{}
\dit{emma}{Lovely! What did you do there?}{}
\dit{jack}{I cooked dinner for them. They loved it!}{}
\end{transcription}
