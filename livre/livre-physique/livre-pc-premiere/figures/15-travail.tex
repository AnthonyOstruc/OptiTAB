% Travail d'une force constante : moteur, nul ou résistant selon l'angle avec le déplacement (L'essentiel, chapitre 15)
\begin{tikzpicture}[x=1cm,y=1cm,
  dep/.style={otInk!70,line width=0.9pt,-{Stealth[length=2mm]}},
  force/.style={otGreenDark,line width=1.2pt,-{Stealth[length=2.4mm]}}]
\foreach \bx/\angl/\titre/\coul in {0/35/{$0\leqslant\alpha<\ang{90}$ : $W>0$, moteur}/otGreenDark,4.6/90/{$\alpha=\ang{90}$ : $W=0$}/otMuted,9.2/145/{$\alpha>\ang{90}$ : $W<0$, résistant}/otRed!80!black}{%
  \begin{scope}[shift={(\bx,0)}]
    \fill[otInk] (0,0) circle (0.05); \node[pcLegende,anchor=north] at (0,-0.05) {A};
    \fill[otInk] (3.0,0) circle (0.05); \node[pcLegende,anchor=north] at (3.0,-0.05) {B};
    \draw[dep] (0,0) -- (3.0,0);
    \draw[force] (1.2,0) -- ++(\angl:1.3) node[pcLegende,anchor=south,text=otGreenDark] {$\vec{F}$};
    \draw[otOrange!85!black,line width=0.5pt] (1.65,0) arc (0:\angl:0.45);
    \node[pcLegende,text=otOrange!85!black] at ({1.2+0.75*cos(\angl/2)},{0.75*sin(\angl/2)}) {$\alpha$};
    \node[pcLegende,anchor=north,text=\coul] at (1.5,-0.45) {\titre};
  \end{scope}}
\end{tikzpicture}
