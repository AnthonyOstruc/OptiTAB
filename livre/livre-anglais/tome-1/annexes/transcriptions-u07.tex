\corrigesUnite{7}{Do you like coffee?}

\begin{pisteaudio}{7.07}{La forme négative (p.~\pageref{piste-7.07})}
\annonce{Listen and repeat.}
\repetepaire{I like tea.}{I don't like tea.}
\repetepaire{She drinks coffee.}{She doesn't drink coffee.}
\repetepaire{We eat meat.}{We don't eat meat.}
\repetepaire{He cooks.}{He doesn't cook.}
\repetepaire{They play music.}{They don't play music.}
\repetepaire{Emma likes coffee.}{Emma doesn't like coffee.}
\end{pisteaudio}

\begin{pisteaudio}{7.08}{Questions et réponses courtes (p.~\pageref{piste-7.08})}
\annonce{Listen and repeat.}
\repetepaire{Do you like jazz?}{Yes, I do.}
\repetepaire{Does she like tea?}{No, she doesn't.}
\repetepaire{Do they eat fish?}{Yes, they do.}
\repetepaire{Does he cook?}{No, he doesn't.}
\repetepaire{What do you drink?}{I drink water.}
\repetepaire{Where does she work?}{She works in New York.}
\end{pisteaudio}

\begin{transcription}{7.11}{Exercice 8 · Qui aime quoi ?}
\repere{First, Emma}
\dire{Emma, do you like coffee?}
\dit{emma}{No, I don't. I like tea. I drink tea every day!}{}
\dire{Do you like cooking?}
\dit{emma}{Yes, I do. I love cooking.}{}
\repere{Now, Jack}
\dire{Jack, do you like tea?}
\dit{jack}{Tea? No, I don't. I love coffee!}{}
\dire{And do you like cooking?}
\dit{jack}{No, I don't. I eat sandwiches.}{}
\repere{And now, Sam}
\dire{Sam, do you like coffee?}
\dit{sam}{Of course! I'm a barista! And I like tea too.}{}
\dire{Do you like cooking?}
\dit{sam}{No, I don't. My sister cooks for me!}{}
\end{transcription}

\begin{transcription}{7.12}{Exercice 9 · Dictée à trous}
\dit{lopez}{Do you like Mexican food, Léa?}{}
\dit{lea}{Yes, I do! I love tacos.}{}
\dit{lopez}{Does your family like spicy food?}{}
\dit{lea}{No, they don't. My mother doesn't like it.}{}
\dit{lopez}{What do you cook in the evening?}{}
\dit{lea}{I cook pasta, or I eat a sandwich!}{}
\dit{lopez}{Come to my place on Sunday. I make great tacos!}{}
\end{transcription}
